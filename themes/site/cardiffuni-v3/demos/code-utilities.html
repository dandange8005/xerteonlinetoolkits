Preformatted Code Block
Multi-line code with preserved formatting

function calculateTotal(items) {
    let total = 0;
    for (let item of items) {
        total += item.price * item.quantity;
    }
    return total;
}

const cart = [
    { name: 'Book', price: 15.99, quantity: 2 },
    { name: 'Pen', price: 2.50, quantity: 5 }
];

console.log(calculateTotal(cart)); // 44.48

Using CSS Variables

CSS custom properties can be used in your stylesheets:

    padding: var(--cu-space-4);
    font-size: var(--cu-text-lead);
    color: var(--cu-ink);
    margin-bottom: var(--cu-space-8);
}
